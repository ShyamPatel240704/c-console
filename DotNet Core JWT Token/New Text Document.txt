            // ✅ Logout Logic
            var action = HttpContext.Request.Query["action"].ToString();

            if (action == "logout")
            {
                HttpContext.Response.Cookies.Delete("jwtToken");
                HttpContext.Response.Redirect("/Admin/LoginForm");
                return Content("");
            }

            // ✅ JWT Claims mathi RegId & Roll lo
            var regIdClaim = HttpContext.User.FindFirst(ClaimTypes.NameIdentifier);
            var rollClaim = HttpContext.User.FindFirst(ClaimTypes.Role);

            if (regIdClaim == null || rollClaim == null)
            {
                return Content(""); // user not logged in
            }

            int regId = Convert.ToInt32(regIdClaim.Value);
            int roll = Convert.ToInt32(rollClaim.Value);

            var model = await db.GetByAdminDataList(regId, roll);

            return View("AdminMenu", model);
